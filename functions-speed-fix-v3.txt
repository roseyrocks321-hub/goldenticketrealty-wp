/**
 * GTR Speed Optimization v3
 * Fixes: Elementor defer, font-display swap, preconnect cleanup, bloat removal
 * Safe for: Desktop + Mobile (no fetchpriority that causes TBT issues)
 * Reversible: Deactivate snippet to restore
 */

// ========== 1. DEFER ELEMENTOR SCRIPTS ==========
add_filter('script_loader_tag', function($tag, $handle) {
    $defer_handles = [
        'elementor-frontend',
        'elementor-pro-frontend',
        'elementor-common',
        'elementor-app-loader',
        'elementor-sticky',
        'elementor-waypoints',
        'elementor-frontend-modules',
    ];
    if (in_array($handle, $defer_handles)) {
        return str_replace(' src=', ' defer src=', $tag);
    }
    return $tag;
}, 10, 2);

// ========== 2. DEFER JQUERY-MIGRATE ==========
add_action('wp_default_scripts', function($scripts) {
    if (!empty($scripts->registered['jquery-migrate'])) {
        $scripts->registered['jquery-migrate']->extra['defer'] = true;
    }
});

// ========== 3. DISABLE BURST STATISTICS FRONTEND JS ==========
add_action('wp_enqueue_scripts', function() {
    wp_dequeue_script('burst');
    wp_dequeue_script('burst-statistics');
    wp_dequeue_script('burst-frontend');
}, 100);

// ========== 4. DISABLE ALT MANAGER FRONTEND JS ==========
add_action('wp_enqueue_scripts', function() {
    wp_dequeue_script('alt-manager');
    wp_dequeue_script('alt-manager-frontend');
}, 100);

// ========== 5. DEFER MSGSNDR FORM SCRIPTS ==========
add_filter('script_loader_tag', function($tag, $handle) {
    if (strpos($handle, 'msgsndr') !== false || strpos($tag, 'msgsndr') !== false) {
        return str_replace(' src=', ' defer src=', $tag);
    }
    return $tag;
}, 10, 2);

// ========== 6. REMOVE WORDPRESS EMOJI DETECTION JS ==========
remove_action('wp_head', 'print_emoji_detection_script', 7);
remove_action('admin_print_scripts', 'print_emoji_detection_script');
remove_action('wp_print_styles', 'print_emoji_styles');
remove_action('admin_print_styles', 'print_emoji_styles');

// ========== 7. DISABLE WP EMBED SCRIPT ==========
add_action('wp_footer', function() {
    wp_dequeue_script('wp-embed');
});

// ========== 8. REMOVE EXTRA PRECONNECT LINKS ==========
add_filter('wp_resource_hints', function($urls, $relation_type) {
    if ($relation_type === 'preconnect' || $relation_type === 'dns-prefetch') {
        return array_filter($urls, function($url) {
            $href = is_array($url) ? ($url['href'] ?? '') : $url;
            return strpos($href, 'fonts.googleapis.com') !== false 
                || strpos($href, 'fonts.gstatic.com') !== false;
        });
    }
    return $urls;
}, 10, 2);

// ========== 9. ADD font-display: swap TO GOOGLE FONTS ==========
add_action('wp_enqueue_scripts', function() {
    global $wp_styles;
    if (!$wp_styles) return;
    foreach ($wp_styles->registered as $handle => $style) {
        if (strpos($style->src, 'fonts.googleapis.com') !== false) {
            if (strpos($style->src, 'display=') === false) {
                $style->src = add_query_arg('display', 'swap', $style->src);
            }
        }
    }
}, 20);
